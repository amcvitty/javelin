1"""What one cell looks like on screen, worked out without a terminal. 

2 

3Four shapes, one per region of the display: a header card of label/value 

4pairs, a row per input slot, a row per known output, and a pair of source 

5strings -- the node as written and as compiled. Each is plain data, so what 

6is shown can be tested without a terminal attached and the terminal library 

7stays on the other side of this module. 

8 

9Nothing here evaluates anything. Reading a cell's value, its slots and its 

10outputs is what the engine already knows; the statically resolvable slots come 

11back filled in because working them out is free, and every other slot reports 

12itself unresolved and waits to be asked. A body runs only when something calls 

13`expand_slot`, and nothing here does. 

14 

15Two questions the display has to keep apart, because the engine does: 

16 

17*unresolved* -- nothing has looked yet, and looking may cost an evaluation. 

18*resolved to no cells* -- something looked, and there is genuinely nothing 

19there: a guard blocked the call site, or a map edge ran over an empty 

20collection. Both come back as an empty tuple of cells, and the value column 

21says which. 

22""" 

23

29#: How much room the identity column has before it is truncated. 

30IDENTITY_WIDTH = 44 

31 

32#: How much room a value has. A slot's value can be a whole collection -- a 

33#: book's list of position paths is one string per member -- and one row must 

34#: not be able to push the rest of the table off the screen. 

35VALUE_WIDTH = 36 

36 

37#: How many of a map edge's elements are shown before the rest are summarised. 

38#: A book of a thousand positions is one slot, and one slot must not be able 

39#: to fill the display. 

40MAP_ROW_CAP = 12 

41 

42#: The input table's columns, in order. 

43INPUT_COLUMNS = ("slot", "kind", "identity", "value", "guard", "reads") 

44 

45#: The output table's columns. Its own schema: an output is a cell, not a slot, 

46#: so it has no kind, no guard and no read set. 

47OUTPUT_COLUMNS = ("cell", "value") 

48 

49_ELLIPSIS = "..." 

50 

51#: A terminal whose value the key does not carry, which `None` would look like. 

52_MISSING = object() 

53 

54#: Shown in a source pane in place of a body neither accessor could read. 

55SOURCE_UNAVAILABLE = "Source unavailable"

58def truncate_left(text, width=IDENTITY_WIDTH): 

59 """Shorten from the left, keeping the tail. 

60 

61 Namespace paths share their heads and differ at their tails: 

62 `/mkt/EQ/ACME/Market` and `/mkt/EQ/WIDGET/Market` are told apart by what 

63 comes last. Cutting the front is what keeps two markets on different 

64 underlyings distinguishable. 

65 """ 

66 if len(text) <= width: 

67 return text 

68 return _ELLIPSIS + text[len(text) - width + len(_ELLIPSIS) :] 

69 

70 

71def truncate_right(text, width): 

72 """Shorten from the right, keeping the head. 

73 

74 The other half of the rule. Source text and values are discriminated by 

75 what comes *first* -- `legs = [...]` is named by `legs`, and a long list is 

76 recognised by its first few elements -- so they lose their tails. Only a 

77 cell identity, whose namespace path shares its head with its neighbours, 

78 is cut the other way round. 

79 """ 

80 if len(text) <= width: 

81 return text 

82 return text[: width - len(_ELLIPSIS)] + _ELLIPSIS

85def object_name(obj): 

86 """What to call an object on screen. 

87 

88 A namespace object knows its own name, which is the useful thing to show. 

89 Anything else is named by its class -- never by its `repr`, which would 

90 put a memory address in front of a reader as though it meant something. 

91 """ 

92 name = getattr(obj, "name", None) 

93 return name if isinstance(name, str) else type(obj).__name__

96def arguments(cell): 

97 """A cell's arguments as written in a call, or an empty string for none.""" 

98 return ", ".join(repr(arg) for arg in cell.args)

101def cell_label(cell, *, relative_to=None): 

102 """One cell, in this repo's own vocabulary: object, method and arguments. 

103 

104 Not `Cell.__str__`, which names the class. On screen the object is the 

105 interesting half -- two positions of the same class are the same string 

106 otherwise. 

107 

108 `relative_to` is the object of the cell currently on screen. A row naming 

109 a cell on that same object is named `self.method(...)` rather than 

110 repeating the path a reader is already looking at -- the identity column 

111 only has to earn its width when the receiver is somewhere else. 

112 """ 

113 same_object = relative_to is not None and cell.obj is relative_to 

114 receiver = "self" if same_object else object_name(cell.obj) 

115 return f"{receiver}.{cell.method_name}({arguments(cell)})"

118def value_text(cell_value, width=VALUE_WIDTH): 

119 """A value narrow enough for a table cell, still carrying its state. 

120 

121 The state is what makes a value safe to show, so it is never what gets 

122 cut: the value itself is shortened and the qualifier kept, rather than 

123 truncating `CellValue`'s own text and losing `(dirty)` off the end of it. 

124 """ 

125 if cell_value.state is ValueState.UNCOMPUTED: 

126 return str(cell_value.state) 

127 shown = truncate_right(repr(cell_value.value), width) 

128 if cell_value.state is ValueState.CLEAN: 

129 return shown 

130 return f"{shown} ({cell_value.state})"

133def value_type(cell_value): 

134 """The type of a value, or blank when there is no value to take it of.""" 

135 if cell_value.state is ValueState.UNCOMPUTED: 

136 return ""

181@dataclass(frozen=True) 

182class InputRow: 

183 """One row of the input table: one slot, or one element of a map edge. 

184 

185 `slot` is the slot's index, sub-indexed as `4.0`, `4.1` for the elements a 

186 map edge named, so an element is always visibly part of its slot. 

187 

188 `slot_index` and `sub_index` are not columns -- nothing displays them -- 

189 but they are how navigation finds its way back to the slot and, for a map 

190 edge, the element this row names. `sub_index` is `None` for a row that 

191 names no single element: a non-map edge, a terminal, a local, or the "N 

192 more" row a capped map edge leaves behind. 

193 """ 

194 

195 slot: str 

196 kind: str 

197 identity: str 

198 value: str 

199 guard: str 

200 reads: str 

201 slot_index: int 

202 sub_index: int | None = None

205def _reads(slot): 

206 """A slot's read set, in slot order so it reads like slot indices.""" 

207 return ", ".join(str(index) for index in sorted(slot.reads))

210def _terminal_values(cell): 

211 """Each parameter's value, by name, taken straight from the key. 

212 

213 A terminal needs no computing -- the key already carries its value -- so 

214 showing it costs nothing. Bound through the node's own signature so that 

215 defaults appear for arguments the call left out. 

216 """ 

217 bound = cell.node.compiled.signature.bind(*cell.args) 

218 bound.apply_defaults() 

219 return bound.arguments

222def _row(slot, identity, value, *, sub=None, nav_sub=None): 

223 """One input row, with the slot index sub-indexed if this is an element. 

224 

225 `value` arrives ready to show -- `value_text` for a cell's value, a plain 

226 word for a slot that has no cell to have one. 

227 

228 `nav_sub` is the element index navigation should use, when it differs from 

229 `sub`'s display purpose -- the "N more" row is labelled with the count it 

230 stands for but names no element of its own, so it leaves `nav_sub` at its 

231 default of `None` rather than passing the count. 

232 """ 

233 index = str(slot.index) if sub is None else f"{slot.index}.{sub}" 

234 return InputRow( 

235 slot=index, 

236 kind=str(slot.kind), 

237 identity=identity, 

238 value=value, 

239 guard=slot.guard_source or "", 

240 reads=_reads(slot), 

241 slot_index=slot.index, 

242 sub_index=nav_sub, 

243 )

246def unresolved_reason(slot): 

247 """Why an edge that resolved to no cells named nothing. 

248 

249 Distinguishing these on screen is the whole reason the engine keeps 

250 "resolved to none" apart from "not yet resolved". Shared with navigation, 

251 so a row's value column and a failed drill's reason say the same thing. 

252 """ 

253 if slot.kind is InputKind.MAP_EDGE: 

254 return "no elements" 

255 return "not reached"

258def _target_call(slot): 

259 """How to name an edge with no cell to name it after. 

260 

261 The method it calls, which is the most of the answer that is known before 

262 the edge is resolved -- and is recognisable, which a map edge's own call 

263 site is not: that is a whole comprehension, and truncating it would throw 

264 away the end a reader needs. 

265 """ 

266 return f"{slot.target}()"

269def _edge_rows(slot, width, cap, relative_to): 

270 """The rows one edge contributes: unresolved, empty, one cell, or many.""" 

271 if slot.cells is UNRESOLVED: 

272 return (_row(slot, _target_call(slot), str(UNRESOLVED)),) 

273 if not slot.cells: 

274 return (_row(slot, _target_call(slot), unresolved_reason(slot)),) 

275 if slot.kind is not InputKind.MAP_EDGE: 

276 cell = slot.cells[0] 

277 label = cell_label(cell, relative_to=relative_to) 

278 return (_row(slot, truncate_left(label, width), value_text(cell.value)),) 

279 

280 rows = [ 

281 _row( 

282 slot, 

283 truncate_left(cell_label(cell, relative_to=relative_to), width), 

284 value_text(cell.value), 

285 sub=sub, 

286 nav_sub=sub, 

287 ) 

288 for sub, cell in enumerate(slot.cells[:cap]) 

289 ] 

290 remainder = len(slot.cells) - cap 

291 if remainder > 0: 

292 rows.append(_row(slot, f"... {remainder} more", "", sub=cap)) 

293 return tuple(rows)

296def input_rows(cell, *, width=IDENTITY_WIDTH, cap=MAP_ROW_CAP): 

297 """A row per slot in `ivs` order, with a map edge's elements fanned out. 

298 

299 Every slot appears, whatever fills it: the table is the `ivs` array, and a 

300 gap in it would be a lie about what the body reads. A slot that names no 

301 cells -- a terminal, a hoisted local -- shows the slot as written. 

302 """ 

303 terminals = _terminal_values(cell) 

304 rows = [] 

305 for slot in cell.slots: 

306 if slot.kind is InputKind.TERMINAL: 

307 value = terminals.get(slot.source, _MISSING) 

308 shown = ( 

309 "" if value is _MISSING else truncate_right(repr(value), VALUE_WIDTH) 

310 ) 

311 rows.append(_row(slot, truncate_right(slot.source, width), shown)) 

312 elif slot.kind is InputKind.LOCAL: 

313 # A local's value would mean running its expression, and looking 

314 # is meant to be free. Its name is at the front of its source, so 

315 # this is the one identity that keeps its head. 

316 rows.append(_row(slot, truncate_right(slot.source, width), "")) 

317 else: 

318 rows.extend(_edge_rows(slot, width, cap, cell.obj)) 

319 return tuple(rows)

322@dataclass(frozen=True) 

323class OutputRow: 

324 """One cell known to read this one. 

325 

326 `target` is not a column -- nothing displays it -- but it is the cell 

327 navigation drills into: an output is a cell already, with nothing to 

328 resolve first. 

329 """ 

330 

331 cell: str 

332 value: str 

333 target: Cell

336def output_rows(cell, *, width=IDENTITY_WIDTH): 

337 """A row per cell known to read this one, in a stable order. 

338 

339 Partial by nature, and deliberately not labelled as such: a cell can 

340 always be added, so "the ones known so far" is the only output list that 

341 could ever exist, and saying so on every screen would say nothing. 

342 """ 

343 entries = sorted( 

344 ( 

345 (cell_label(output, relative_to=cell.obj), value_text(output.value), output) 

346 for output in cell.outputs 

347 ), 

348 key=lambda entry: entry[:2], 

349 ) 

350 return tuple( 

351 OutputRow(cell=truncate_left(label, width), value=value, target=target) 

352 for label, value, target in entries 

353 )
