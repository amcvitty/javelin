1"""Rewriting a node's body into a pure function of its input values. 

2 

3Every parameter and every `self.<node>()` call becomes an `ivs[i]` subscript, 

4and each one yields an input saying how it is produced. A node call also 

5records the *guard* under which its call site is reached, so an input that a 

6particular invocation never touches is never made a dependency -- which is 

7what stops a recursive node expanding forever. 

8 

9A comprehension whose element is a node call is rewritten the same way: the 

10whole comprehension becomes one `ivs[i]` holding the list of results, and the 

11input it yields is a `MapEdge` naming one cell per element. 

12"""

75def binding_counts(func_def): 

76 """How many times each name is bound in the body. 

77 

78 Anything not in `Load` context is a binding: assignment targets, loop 

79 variables, `with ... as`, comprehension and lambda parameters. The names are 

80 what a hoisted input cannot see; a name bound exactly once, by a plain 

81 top-level assignment, is a candidate for hoisting. 

82 """ 

83 counts = {} 

84 for n in ast.walk(func_def): 

85 if isinstance(n, ast.Name) and not isinstance(n.ctx, ast.Load): 

86 counts[n.id] = counts.get(n.id, 0) + 1 

87 elif isinstance(n, (ast.Lambda, ast.FunctionDef)) and n is not func_def: 87 ↛ 88line 87 didn't jump to line 88 because the condition on line 87 was never true

88 for a in ast.walk(n.args): 

89 if isinstance(a, ast.arg): 

90 counts[a.arg] = counts.get(a.arg, 0) + 1 

91 return counts

346 def _call_kind(self, func_expr: ast.Attribute, elements=()): 

347 """What a call site is, from the shape of the attribute being called. 

348 

349 "input" -- an input: self.<node>(...), or a call on an object the 

350 graph produced, like self.Other().value(). Whether the 

351 latter names a node is only knowable once the receiver has 

352 been evaluated, so the runtime decides. 

353 "plain" -- self.<method>(...) where the method is not a node. 

354 None -- nothing to do with the graph. 

355 

356 `elements` are the loop variables in force, if we are looking inside a 

357 comprehension: a call on an element is an input for the same reason a 

358 call on another cell's value is. 

359 """ 

360 if self._is_self_receiver(func_expr): 

361 attr = getattr(self.owner, func_expr.attr, None) 

362 return "input" if self.is_node(attr) else "plain" 

363 if self._reads_self(func_expr.value): 

364 return "input" 

365 if any(self._reads_name(func_expr.value, name) for name in elements): 

366 return "input" 

367 return None

369 def _check_hoistable(self, expr, source, allow=()): 

370 """Whether an expression may be lifted above the body. 

371 

372 `allow` names locals it may read anyway -- a map's loop variable, which 

373 is a parameter of the lambdas it is compiled into rather than a local 

374 the body binds. 

375 """ 

376 for n in ast.walk(expr): 

377 if ( 

378 isinstance(n, ast.Name) 

379 and n.id in self.locals 

380 and n.id not in self.hoisted 

381 and n.id not in allow 

382 ): 

383 raise ValueError( 

384 f"input {source!r} refers to local {n.id!r}; an input may " 

385 "only use the node's parameters, earlier inputs and names " 

386 "from outside the method" 

387 )

400 def _guard_at(self, source): 

401 """The guard in force at a call site, as (expression to compile, text).

440 def visit_Call(self, node): 

441 func = node.func 

442 if not isinstance(func, ast.Attribute): 

443 # Only a call on an attribute can name a node. 

444 return self.generic_visit(node) 

445 kind = self._call_kind(func) 

446 if kind is None: 

447 return self.generic_visit(node) 

448 if kind == "plain": 

449 # A plain method call stays in the body; only its arguments are 

450 # rewritten. 

451 return self._visit_arguments(node) 

452 

453 source = ast.unparse(node) 

454 if self.repeat: 

455 raise ValueError( 

456 f"{source} is inside a loop, comprehension or lambda, so it " 

457 "cannot be a single input" 

458 ) 

459 if any(isinstance(a, ast.Starred) for a in node.args) or any( 459 ↛ 462line 459 didn't jump to line 462 because the condition on line 459 was never true

460 k.arg is None for k in node.keywords 

461 ): 

462 raise ValueError(f"{source}: *args and **kwargs are not supported") 

463 

464 # The object being called is resolved first, so a node call inside the 

465 # receiver becomes an earlier input that this one refers to. 

466 if self._is_self_receiver(func): 

467 receiver = ast.Name(id=self.self_name, ctx=ast.Load()) 

468 else: 

469 receiver = self.visit(func.value) 

470 # Node calls nested in the arguments become inputs too. 

471 node = self._visit_arguments(node) 

472 for expr in [receiver, *node.args, *(k.value for k in node.keywords)]: 

473 self._check_hoistable(expr, source) 

474 

475 guard, guard_source = self._guard_at(source) 

476 

477 index = self._alloc() 

478 self.edges.append( 

479 RawCallEdge( 

480 index=index, 

481 target=func.attr, 

482 guard=guard, 

483 guard_source=guard_source, 

484 source=source, 

485 receiver=receiver, 

486 args=node.args, 

487 keywords=node.keywords, 

488 ) 

489 ) 

490 return ast.copy_location(self._ivs(index), node)

522 def _rejected_comprehension(self, node, reason): 

523 """A comprehension shape we will not map. 

524 

525 Only an error if the *repeated* part of it reaches the graph: the 

526 outermost iterable runs once however many elements there are, so a node 

527 call there is an ordinary edge whatever the rest of the shape is. 

528 """ 

529 source = ast.unparse(node) 

530 node.generators[0].iter = self.visit(node.generators[0].iter) 

531 self.repeat += 1 

532 try: 

533 if self._node_calls(node, self._loop_vars(node)): 

534 raise ValueError(f"{source}: {reason}") 

535 return super().generic_visit(node) 

536 finally: 

537 self.repeat -= 1

553 def _visit_comprehension(self, node): 

554 """Rewrite `[t.pv() for t in <iterable>]` into one `MapEdge`. 

555 

556 The iterable is visited first and outside the repeat guard: it runs once 

557 however many elements there are, so a node call in it is an ordinary 

558 edge, and it has to be an *earlier* input than the map that reads it. 

559 """ 

560 if len(node.generators) != 1: 

561 return self._rejected_comprehension( 

562 node, "only one 'for' clause is supported; give each level its own node" 

563 ) 

564 generator = node.generators[0] 

565 if generator.is_async: 565 ↛ 566line 565 didn't jump to line 566 because the condition on line 565 was never true

566 return self._rejected_comprehension(node, "async is not modelled") 

567 if generator.ifs: 

568 return self._rejected_comprehension( 

569 node, 

570 "an 'if' filter is not supported: it would change how many " 

571 "cells the comprehension names", 

572 ) 

573 if not isinstance(generator.target, ast.Name): 573 ↛ 574line 573 didn't jump to line 574 because the condition on line 573 was never true

574 return self._rejected_comprehension( 

575 node, "the loop variable must be a single name" 

576 ) 

577 

578 source = ast.unparse(node) 

579 var = generator.target.id 

580 generator.iter = self.visit(generator.iter) 

581 calls = self._node_calls(node.elt, (var,)) 

582 if not calls: 582 ↛ 585line 582 didn't jump to line 585 because the condition on line 582 was never true

583 # Nothing in the element touches the graph, so it stays plain code 

584 # over whatever the iterable produced. 

585 return self._plain_element(node) 

586 if len(calls) > 1: 

587 raise ValueError( 

588 f"{source}: a comprehension may contain one node call, not " 

589 f"{len(calls)}; give {var!r} a node that combines them, or " 

590 "hoist a loop-invariant call into an assignment above" 

591 ) 

592 return self._map_edge(node, calls[0], generator, var, source)

594 def _map_edge(self, node, call, generator, var, source): 

595 if self.repeat: 595 ↛ 596line 595 didn't jump to line 596 because the condition on line 595 was never true

596 raise ValueError( 

597 f"{source} is inside a loop, comprehension or lambda, so it " 

598 "cannot be a single input" 

599 ) 

600 if call is not node.elt: 

601 raise ValueError( 

602 f"{source}: the element must be the node call itself, not an " 

603 f"expression around it; put the rest on a node of {var!r}" 

604 ) 

605 if var in (self.self_name, "node", "ivs"): 605 ↛ 606line 605 didn't jump to line 606 because the condition on line 605 was never true

606 raise ValueError( 

607 f"{source}: {var!r} is not available as a comprehension variable" 

608 ) 

609 func = call.func 

610 assert isinstance(func, ast.Attribute) 

611 if any(isinstance(a, ast.Starred) for a in call.args) or any( 611 ↛ 614line 611 didn't jump to line 614 because the condition on line 611 was never true

612 k.arg is None for k in call.keywords 

613 ): 

614 raise ValueError(f"{source}: *args and **kwargs are not supported") 

615 

616 self._check_hoistable(generator.iter, source) 

617 # The receiver and the arguments may mention the element -- they are 

618 # compiled with it as a parameter. They may not contain a node call of 

619 # their own, but nothing needs checking here: such a call would have 

620 # been a second candidate above, and one is the limit. 

621 receiver = ( 

622 ast.Name(id=self.self_name, ctx=ast.Load()) 

623 if self._is_self_receiver(func) 

624 else self.visit(func.value) 

625 ) 

626 call = self._visit_arguments(call) 

627 for expr in [receiver, *call.args, *(k.value for k in call.keywords)]: 

628 self._check_hoistable(expr, source, allow=(var,)) 

629

631 

632 index = self._alloc() 

633 self.edges.append( 

634 RawMapEdge( 

635 index=index, 

636 target=func.attr, 

637 guard=guard, 

638 guard_source=guard_source, 

639 source=source, 

640 over=generator.iter, 

641 receiver=receiver, 

642 args=call.args, 

643 keywords=call.keywords, 

644 var=var, 

645 ) 

646 ) 

647 return ast.copy_location(self._ivs(index), node)

651 self.repeat += 1 

652 try:
