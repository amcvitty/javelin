1"""Walking the graph: a focused cell, a back-stack, and a depth. 

2 

3None of this exists in the graph. A focused cell, the path taken to reach it, 

4and how deep that path runs are concepts that exist only because a human is 

5looking at a terminal -- the graph describes cells and knows nothing about who 

6is reading them. So this module imports no terminal library and knows nothing 

7of tables or key bindings either; it is plain state, testable directly. 

8 

9Drilling and resolving never run a node's own body. `expand_slot` may still 

10evaluate whatever an edge's guard or a map's collection reads -- the cost its 

11`blocked_by` warned about -- but the cell being navigated to is never the one 

12evaluated. Only `evaluate` is deliberate about running bodies, and nothing 

13here calls it except the two methods named for it. 

14 

15The back-stack is a history, not a set: revisiting a cell already on it is 

16ordinary, and going back at the root does nothing rather than erroring. 

17""" 

18 

19from dataclasses import dataclass 

20 

21from graph import UNRESOLVED, Cell, InputKind

23from .render import unresolved_reason

25#: Why a slot of this kind has no cell behind it -- it was never an edge, so 

26#: there was never anything to resolve. 

27_NOT_AN_EDGE = { 

28 InputKind.TERMINAL: "a terminal has no cell behind it", 

29 InputKind.LOCAL: "a hoisted local has no cell behind it", 

30} 

31 

32#: Why drilling into an unresolved or capped map-edge row does not navigate: 

33#: it may now name several cells, and one keystroke cannot pick among them. 

34_PICK_AN_ELEMENT = "a map edge -- drill into one of its elements" 

35 

36#: Why evaluating a row that has not been resolved does not run anything: 

37#: evaluation is deliberate and is never a side effect of looking. 

38_NOT_YET_RESOLVED = "not yet resolved" 

39

41@dataclass(frozen=True) 

42class ActionResult: 

43 """What came of trying to act on a row. 

44 

45 `ok` is the whole answer a caller needs to act on -- whether to move on or 

46 stay put. `reason` is only for a human, filled in when there was nothing 

47 to act on. 

48 """ 

49 

50 ok: bool 

51 reason: str = "" 

52

54class Navigation: 

55 """Where a human is in the graph, and the path taken to get there. 

56 

57 `focus` is the cell currently on screen. `depth` and `breadcrumb` fall out 

58 of the back-stack rather than being tracked alongside it, so they can 

59 never drift out of step with the path actually taken. 

60 """ 

61 

62 def __init__(self, root: Cell): 

63 self.focus = root 

64 self._stack: list[Cell] = [] 

65 

66 @property 

67 def depth(self) -> int: 

68 """How many cells below the root the current focus sits.""" 

69 return len(self._stack) 

70 

71 @property 

72 def breadcrumb(self) -> tuple[Cell, ...]: 

73 """The path taken to the current focus, root first.""" 

74 return (*self._stack, self.focus) 

75 

76 def back(self) -> None: 

77 """Return to the previous cell. Does nothing at the root.""" 

78 if self._stack: 

79 self.focus = self._stack.pop() 

80 

81 def resolve_slot(self, index: int) -> None: 

82 """Fill in one of the focused cell's slots, leaving focus where it is.""" 

83 self.focus.expand_slot(index) 

84 

85 def drill_slot(self, index: int, sub: int | None = None) -> ActionResult: 

86 """Navigate to the cell one of the focused cell's slots names. 

87 

88 Resolves the slot first if nothing has looked yet -- one keystroke for 

89 one intention. A slot with no single cell to land on does not 

90 navigate: a terminal or hoisted local never had one, a blocked call 

91 site or an empty map resolved to none, and an unresolved or capped map 

92 edge may now name several, which one keystroke cannot choose among. 

93 """ 

94 slot = self.focus.slots[index] 

95 not_an_edge = _NOT_AN_EDGE.get(slot.kind) 

96 if not_an_edge is not None: 

97 return ActionResult(False, not_an_edge) 

98 cells = slot.cells 

99 if cells is UNRESOLVED: 

100 cells = self.focus.expand_slot(index) 

101 if not isinstance(cells, tuple) or not cells: 

102 return ActionResult(False, unresolved_reason(slot)) 

103 if slot.kind is InputKind.MAP_EDGE and sub is None: 

104 return ActionResult(False, _PICK_AN_ELEMENT) 

105 self._navigate(cells[sub if sub is not None else 0]) 

106 return ActionResult(True) 

107 

108 def drill_output(self, cell: Cell) -> ActionResult: 

109 """Navigate to a cell already known to read the focused one. 

110 

111 Always succeeds: an output is a cell, resolved by definition, so 

112 drilling into one works exactly as drilling into an input does. 

113 """ 

114 self._navigate(cell) 

115 return ActionResult(True) 

116 

117 def evaluate_slot(self, index: int, sub: int | None = None) -> ActionResult: 

118 """Run the body of the cell one of the focused cell's slots names. 

119 

120 Unlike drilling, this never resolves first: evaluation is deliberate, 

121 so a row nothing has looked at yet is left exactly as unresolved as it 

122 was, rather than being resolved as a side effect of trying. 

123 """ 

124 slot = self.focus.slots[index] 

125 not_an_edge = _NOT_AN_EDGE.get(slot.kind)

127 return ActionResult(False, not_an_edge) 

128 cells = slot.cells 

129 if cells is UNRESOLVED: 

130 return ActionResult(False, _NOT_YET_RESOLVED) 

131 if not isinstance(cells, tuple) or not cells: 131 ↛ 132line 131 didn't jump to line 132 because the condition on line 131 was never true

133 if slot.kind is InputKind.MAP_EDGE and sub is None: 

134 return ActionResult(False, _PICK_AN_ELEMENT) 

135 cells[sub if sub is not None else 0].evaluate() 

136 return ActionResult(True)

138 def evaluate_output(self, cell: Cell) -> None: 

139 """Run the body of a cell already known to read the focused one. 

140 

141 No `ActionResult`, unlike `evaluate_slot`: an output is a cell 

142 already, the same way `drill_output` needs no resolving-first check, 

143 so there is no refusal for a caller to act on. 

144 """ 

145 cell.evaluate() 

146 

147 def evaluate_focus(self) -> None: 

148 """Run the focused cell's own body, filling in its value and type. 

149 

150 No `ActionResult` either, for the same reason: the focus is always a 

151 cell, never a row that might have nothing behind it. 

152 """ 

153 self.focus.evaluate() 

154 

155 def _navigate(self, cell: Cell) -> None: 

156 self._stack.append(self.focus) 

157 self.focus = cell
